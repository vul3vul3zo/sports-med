/** * 運動防護物資管理系統 - 絕對防呆過濾版 (Code.gs) */ function appendRecord(data) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheets()[0]; // 嚴格防呆過濾：如果是空的、未指定的、或沒有品項名稱，絕對不准寫入試算表！ var itemName = data.itemName ? String(data.itemName).trim() : ''; if (!itemName || itemName === '未指定品項' || itemName === 'undefined' || itemName === 'null') { return { status: "ignored", message: "Skipped empty or unspecified item." }; } // 若第一列沒有標題，自動建立標準表頭 if (sheet.getLastRow() === 0) { sheet.appendRow([ "登記時間", "活動專案", "借領單位", "登記人員", "動作類型", "品項名稱", "數量", "單位", "備註說明" ]); } var timeStr = data.time || Utilities.formatDate(new Date(), "Asia/Taipei", "yyyy/MM/dd HH:mm:ss"); var eventName = data.eventName || '隨隊活動'; var unit = data.unit || '現場組別'; var user = data.user || '現場防護員'; var actionType = data.actionType || '領出'; var qty = Number(data.qty) || 1; var unitName = data.unitName || '件'; var note = data.note || ''; // 正常寫入有效資料 sheet.appendRow([ timeStr, eventName, unit, user, actionType, itemName, qty, unitName, note ]); SpreadsheetApp.flush(); return { status: "success", row: sheet.getLastRow(), time: timeStr }; } function doGet(e) { try { var params = (e && e.parameter) ? e.parameter : {}; // Support quick write through GET if (params.action === 'add' || (params.itemName && params.qty)) { var writeResult = appendRecord(params); if (params.callback) { return ContentService.createTextOutput(params.callback + '(' + JSON.stringify(writeResult) + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(JSON.stringify(writeResult)) .setMimeType(ContentService.MimeType.JSON); } var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheets()[0]; var rows = sheet.getDataRange().getValues(); var resultList = []; // Parse records from sheet for (var i = 1; i < rows.length; i++) { var row = rows[i]; if (!row[0] || !row[5] || String(row[5]) === '未指定品項') continue; resultList.push({ time: row[0] instanceof Date ? Utilities.formatDate(row[0], "Asia/Taipei", "yyyy/MM/dd HH:mm:ss") : String(row[0]), eventName: String(row[1] || ''), unit: String(row[2] || ''), user: String(row[3] || ''), actionType: String(row[4] || ''), itemName: String(row[5] || ''), qty: Number(row[6]) || 0, unitName: String(row[7] || ''), note: String(row[8] || '') }); } var jsonString = JSON.stringify({ status: "success", data: resultList }); var callback = params.callback; // Return JSONP for cross-origin frontend requests if (callback) { return ContentService.createTextOutput(callback + '(' + jsonString + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(jsonString) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { var data = {}; if (e && e.postData && e.postData.contents) { try { data = JSON.parse(e.postData.contents); } catch (err) { data = (e && e.parameter) ? e.parameter : {}; } } else if (e && e.parameter) { data = e.parameter; } var result = appendRecord(data); return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
